Upload this sketch to your Arduino Nano or Uno (disconnect HC-05 RX pin while uploading).

#include <SoftwareSerial.h>

SoftwareSerial bt(10, 11); // RX = D10 (to HC-05 TXD), TX = D11 (to HC-05 RXD via divider)

const int BUZZER = 8;
const int MOTOR  = 9;
const int LED    = 7;

bool ringing = false;
bool pulse   = false;
unsigned long lastToggle = 0;
unsigned long startTime  = 0;

void allOff() {
  digitalWrite(BUZZER, LOW);
  digitalWrite(MOTOR, LOW);
  digitalWrite(LED, LOW);
}

void setup() {
  pinMode(BUZZER, OUTPUT);
  pinMode(MOTOR, OUTPUT);
  pinMode(LED, OUTPUT);
  allOff();
  bt.begin(9600);
}

void loop() {
  if (bt.available()) {
    char c = bt.read();
    if (c == 'R') { ringing = true; startTime = millis(); }
    if (c == 'S') { ringing = false; allOff(); }
  }

  if (ringing) {
    // Auto shut-off after 30 seconds if unclaimed
    if (millis() - startTime > 30000UL) {
      ringing = false;
      allOff();
    } else if (millis() - lastToggle > 300) {
      lastToggle = millis();
      pulse = !pulse;
      digitalWrite(BUZZER, pulse);
      digitalWrite(MOTOR, pulse);
      digitalWrite(LED, pulse);
    }
  }
}

Connect your components on a mini breadboard using the pin map below. Default HC-05 pairing PIN is 1234.

Component Pin Arduino Pin Notes & Protection
HC-05 VCC / GND 5V / GND Main module power
HC-05 TXD D10 Bluetooth TX to Arduino SoftwareSerial RX
HC-05 RXD D11 (via divider) D11 → 1kΩ → RXD; RXD → 2kΩ → GND (3.3V safe logic)
Active Buzzer (+ / -) D8 / GND 5V continuous-tone active buzzer
Red Alert LED (+ / -) D7 (via 220Ω) / GND Long leg (+) to 220Ω resistor
2N2222 Transistor Base: D9 (1kΩ), Emitter: GND Drives the high-current vibration motor safely
Coin Vibration Motor 5V & Collector Add 1N4007 flyback diode across motor (stripe toward 5V)

// GLOBAL VARIABLES
initialize global cartList to [ create empty list ]
initialize global totalAmount to 0
initialize global orderNumber to 100

// ADD ITEM TO CART (Repeat for Burger ₱65, Chicken ₱95, Combos, Fries, etc.)
when BtnAddBurger.Click do:
    add items to list (get global cartList, item: "Sir Jake Burger - ₱65")
    set global totalAmount to (get global totalAmount + 65)
    set BtnViewCart.Text to join("VIEW CART (", length of list(get global cartList), ")")

// OPEN CART VIEW
when BtnViewCart.Click do:
    set MenuView.Visible to false
    set CartView.Visible to true
    set ListCart.Elements to (get global cartList)
    set LblTotal.Text to join("Total: ₱", get global totalAmount)

// PROCEED TO PAYMENT
when BtnProceedPay.Click do:
    set CartView.Visible to false
    set PayView.Visible to true
    set global orderNumber to (get global orderNumber + 1)
    set LblOrderNo.Text to join("Order #", get global orderNumber)

// SEND ORDER TO CLOUDDB
when BtnConfirmOrder.Click do:
    call CloudDB1.AppendValueToList(
        tag: "active_orders",
        valueToAdd: join("Order #", get global orderNumber, ": ", get global cartList)
    )
    set BtnConfirmOrder.Enabled to false

// RESET FOR NEXT CUSTOMER
when BtnNewOrder.Click do:
    set global cartList to [ create empty list ]
    set global totalAmount to 0
    set BtnViewCart.Text to "VIEW CART (0)"
    set BtnConfirmOrder.Enabled to true
    set PayView.Visible to false
    set MenuView.Visible to true

// BLUETOOTH HC-05 PAIRING & CONNECTION
when ListPickerBT.BeforePicking do:
    set ListPickerBT.Elements to (BluetoothClient1.AddressesAndNames)

when ListPickerBT.AfterPicking do:
    if (call BluetoothClient1.Connect(address: ListPickerBT.Selection)) then:
        set LblBTStatus.Text to "Connected to HC-05 Disc"
    else:
        set LblBTStatus.Text to "Connection Failed"

// REAL-TIME CLOUDDB SYNC
when Screen1.Initialize do:
    call CloudDB1.GetValue(tag: "active_orders", valueIfTagNotThere: [ create empty list ])

when CloudDB1.GotValue(tag, value) do:
    if (get tag = "active_orders") then:
        set ListOrders.Elements to (get value)

when CloudDB1.DataChanged(tag, value) do:
    if (get tag = "active_orders") then:
        set ListOrders.Elements to (get value)

// RING BUZZER DISC ('R') & STOP / CLAIM ('S')
when BtnRing.Click do:
    if (BluetoothClient1.IsConnected) then:
        call BluetoothClient1.SendText(text: "R")

when BtnStop.Click do:
    if (BluetoothClient1.IsConnected) then:
        call BluetoothClient1.SendText(text: "S")
    if (ListOrders.SelectionIndex > 0) then:
        call CloudDB1.RemoveFirstFromList(tag: "active_orders")
